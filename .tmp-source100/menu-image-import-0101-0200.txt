      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    })
    .slice(0, 100);

  return {
    items,
    warnings: (Array.isArray(raw.warnings) ? raw.warnings : [])
      .map((warning) => sanitizeText(warning, 240))
      .filter(Boolean)
      .slice(0, 20),
  };
}
